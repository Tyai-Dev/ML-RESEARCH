\begin{abstract}
This note is a self-contained reference for the Bernoulli
distribution: its definition and elementary properties, the sampling
properties of its natural estimator, maximum likelihood and maximum a
posteriori estimation, and the numerical solution of the maximum
likelihood problem by stochastic gradient descent. All quantitative
statements are verified numerically by the companion program
\texttt{bernoulli.py}.
\end{abstract}

\section{The distribution}

\begin{definition}
A random variable $X$ follows the \emph{Bernoulli distribution} with
parameter $p \in [0,1]$, written $X \sim \mathrm{Bernoulli}(p)$, if
$X$ takes values in $\{0, 1\}$ with
\begin{equation}
  \Pr(X = 1) = p, \qquad \Pr(X = 0) = 1 - p.
\end{equation}
\end{definition}

The probability mass function admits the compact form
\begin{equation}
  f(x; p) \;=\; p^{x}(1-p)^{1-x}, \qquad x \in \{0, 1\},
  \label{eq:pmf}
\end{equation}
and the cumulative distribution function is the step function
\begin{equation}
  F(x; p) \;=\;
  \begin{cases}
    0, & x < 0,\\
    1 - p, & 0 \le x < 1,\\
    1, & x \ge 1.
  \end{cases}
  \label{eq:cdf}
\end{equation}

\begin{proposition}[Exponential family form]
\label{prop:expfam}
The pmf \eqref{eq:pmf} is a full-rank one-parameter exponential
family: with natural parameter $\theta = \log\frac{p}{1-p}$ and
log-partition $A(\theta) = \log(1 + e^{\theta})$,
\begin{equation}
  f(x; p) \;=\; \exp\bigl(x\theta - A(\theta)\bigr),
  \qquad
  A'(\theta) = \sigma(\theta) = p,
  \qquad
  A''(\theta) = p(1-p):
  \label{eq:expfam}
\end{equation}
the derivatives of the log-partition generate the mean and variance.
\end{proposition}

\section{Statistics of the estimator}

Let $x_1, \dots, x_n$ be an i.i.d.\ sample from
$\mathrm{Bernoulli}(p)$ and write $m = \sum_i x_i$ for the number of
successes.

\begin{proposition}[Sufficiency]
The likelihood depends on the data only through $m$:
\begin{equation}
  L(p) \;=\; \prod_{i=1}^{n} p^{x_i}(1-p)^{1-x_i}
        \;=\; p^{m}(1-p)^{n-m}.
  \label{eq:lik}
\end{equation}
By the factorization criterion, $m$ (equivalently $\xbar = m/n$) is a
sufficient statistic for $p$.
\end{proposition}

Throughout, $\phat = m/n$ denotes the sample mean; its sampling
properties follow from $m \sim \mathrm{Binomial}(n, p)$.

\begin{proposition}[Finite-sample properties]
\label{prop:finite}
For every $n \ge 1$,
\begin{equation}
  \E[\phat] = p,
  \qquad
  \Var(\phat) = \frac{p(1-p)}{n},
  \qquad
  \mathrm{MSE}(\phat) = \frac{p(1-p)}{n}.
\end{equation}
Moreover $\Var(\phat)$ attains the Cram\'er--Rao lower bound
$1/(n I(p))$: among unbiased estimators of $p$, the sample mean is
\emph{efficient} (see \textnormal{Theory/statistics} for the general
bound).
\end{proposition}

\begin{proposition}[Asymptotics]
\label{prop:asym}
$\phat \xrightarrow{\,a.s.\,} p$ (strong law of large numbers), and by
the central limit theorem
\begin{equation}
  \sqrt{n}\,(\phat - p) \;\xrightarrow{\ d\ }\;
  \mathcal{N}\bigl(0,\, p(1-p)\bigr).
  \label{eq:clt}
\end{equation}
\end{proposition}

\begin{corollary}[Confidence interval]
An asymptotic $95\%$ confidence interval for $p$ is
$\phat \pm 1.96\sqrt{\phat(1-\phat)/n}$.
\end{corollary}

\begin{theorem}[UMVUE]
\label{thm:umvue}
$\phat = \xbar$ is the unique uniformly minimum-variance unbiased
estimator of $p$.
\end{theorem}

\begin{proof}[Proof sketch]
As a full-rank exponential family (Proposition~\ref{prop:expfam}),
the sufficient statistic $m$ is also \emph{complete}: no nonzero
function of $m$ has expectation zero for all $p$ (a polynomial in $p$
of degree $\le n$ vanishing on $(0,1)$ has all coefficients zero).
Since $\phat = m/n$ is unbiased and a function of the complete
sufficient statistic, the Lehmann--Scheff\'e theorem yields the claim.
\end{proof}

\begin{corollary}[Asymptotic efficiency]
By \eqref{eq:clt} the limiting variance of $\sqrt{n}(\phat - p)$ is
$p(1-p) = 1/I(p)$: the estimator is asymptotically efficient, its
precision fully accounted for by the Fisher information
\eqref{eq:fisher}.
\end{corollary}

\begin{remark}[What is estimable at all]
An unbiased estimator of $g(p)$ from $n$ observations exists if and
only if $g$ is a polynomial of degree at most $n$: the expectation of
any statistic is such a polynomial. Consequently quantities such as
the odds $p/(1-p)$ or $1/p$ admit \emph{no} unbiased estimator at any
sample size --- unbiasedness is a property of the target, not only of
the method. (The MLE handles them by invariance:
the MLE of $g(p)$ is $g(\phat)$, at the price of bias of order
$1/n$.)
\end{remark}

\section{Maximum likelihood and maximum a posteriori estimation}

\subsection{Maximum likelihood}

\begin{theorem}[MLE]
\label{thm:mle}
Let $0 < m < n$. The unique maximizer of \eqref{eq:lik} over
$p \in (0,1)$ is
\begin{equation}
  \boxed{\;\phat_{\mathrm{MLE}} \;=\; \frac{m}{n} \;=\; \xbar.\;}
  \label{eq:mle}
\end{equation}
\end{theorem}

\begin{proof}
Since $\log$ is strictly increasing, maximizing $L$ is equivalent to
maximizing
\begin{equation}
  \ell(p) \;=\; \log L(p) \;=\; m \log p + (n - m)\log(1 - p).
  \label{eq:loglik}
\end{equation}
Differentiating and equating to zero,
\begin{equation*}
  \ell'(p) = \frac{m}{p} - \frac{n-m}{1-p} = 0
  \;\Longleftrightarrow\;
  m(1-p) = (n-m)\,p
  \;\Longleftrightarrow\;
  m = np,
\end{equation*}
which yields \eqref{eq:mle}. Uniqueness follows from strict concavity:
$\ell''(p) = -m/p^{2} - (n-m)/(1-p)^{2} < 0$ on $(0,1)$.
\end{proof}

\begin{remark}[Boundary cases]
If $m = 0$ or $m = n$, the likelihood is monotone on $(0,1)$ and its
supremum is attained at the boundary point $m/n$; formula
\eqref{eq:mle} remains valid.
\end{remark}

\subsection{Maximum a posteriori}

\begin{proposition}[Conjugacy]
Let $p \sim \mathrm{Beta}(\alpha, \beta)$, i.e.\
$\pi(p) \propto p^{\alpha-1}(1-p)^{\beta-1}$. Then the posterior given
the sample is again Beta:
\begin{equation}
  \pi(p \mid x_{1:n}) \;\propto\; L(p)\,\pi(p)
  \;\propto\; p^{\,m+\alpha-1}(1-p)^{\,n-m+\beta-1}
  \;=\; \mathrm{Beta}(m + \alpha,\; n - m + \beta).
\end{equation}
\end{proposition}

\begin{theorem}[MAP]
For $m + \alpha > 1$ and $n - m + \beta > 1$, the posterior mode is
\begin{equation}
  \boxed{\;\phat_{\mathrm{MAP}}
  \;=\; \frac{m + \alpha - 1}{\,n + \alpha + \beta - 2\,}.\;}
  \label{eq:map}
\end{equation}
\end{theorem}

\begin{proof}
The log-posterior is, up to an additive constant not involving $p$,
\begin{equation*}
  \log \pi(p \mid x_{1:n})
  \;=\; (m + \alpha - 1)\log p \;+\; (n - m + \beta - 1)\log(1 - p)
  \;+\; C.
\end{equation*}
Differentiating and equating to zero,
\begin{equation*}
  \frac{\partial}{\partial p}\log \pi(p \mid x_{1:n})
  \;=\; \frac{m + \alpha - 1}{p} - \frac{n - m + \beta - 1}{1 - p}
  \;=\; 0,
\end{equation*}
so $(m + \alpha - 1)(1 - p) = (n - m + \beta - 1)\,p$, i.e.\
$m + \alpha - 1 = p\,(n + \alpha + \beta - 2)$, which is
\eqref{eq:map}. The second derivative
$-(m{+}\alpha{-}1)/p^2 - (n{-}m{+}\beta{-}1)/(1{-}p)^2$ is negative
under the stated conditions, so the stationary point is the unique
maximum --- the same concavity argument as Theorem~\ref{thm:mle},
with the prior's pseudo-counts joining the data's.
\end{proof}

\begin{remark}[Interpretation and limits]
The prior acts as \emph{pseudo-counts}: $\alpha - 1$ successes and
$\beta - 1$ failures observed before any data. In particular
(i) the flat prior $\alpha = \beta = 1$ gives
$\phat_{\mathrm{MAP}} = \phat_{\mathrm{MLE}}$ exactly;
(ii) $\phat_{\mathrm{MAP}} \to \phat_{\mathrm{MLE}}$ as
$n \to \infty$; and (iii) for small $n$ the estimator shrinks toward
the prior mode, trading bias for variance --- numerically, at
$n = 20$ with $\mathrm{Beta}(3,3)$ the MAP mean squared error is
$0.0085$ against the MLE's $0.0105$ ($2 \times 10^4$ replications).
Additive (Laplace) smoothing of categorical counts is precisely the
multinoulli analog of \eqref{eq:map} under a Dirichlet prior.
\end{remark}

\section{Information theory}

\begin{definition}[Score and Fisher information]
The \emph{score} is the derivative of the log-likelihood of one
observation,
\begin{equation}
  s(p; x) \;=\; \frac{\partial}{\partial p} \log f(x; p)
  \;=\; \frac{x - p}{p(1-p)},
  \label{eq:score}
\end{equation}
and the \emph{Fisher information} is its variance. The two classical
identities hold:
\begin{equation}
  \E[s(p; X)] = 0,
  \qquad
  I(p) \;=\; \Var\bigl(s(p; X)\bigr)
        \;=\; -\E\Bigl[\tfrac{\partial^2}{\partial p^2}
              \log f(X; p)\Bigr]
        \;=\; \frac{1}{p(1-p)}.
  \label{eq:fisher}
\end{equation}
Information is largest at the certain ends and smallest at
$p = \tfrac12$: an observation is most informative where outcomes are
most surprising.
\end{definition}

\begin{definition}[Entropy, cross-entropy, KL divergence]
\begin{equation}
  H(p) = -p\log p - (1-p)\log(1-p),
  \qquad
  H(p, q) = -p\log q - (1-p)\log(1-q),
\end{equation}
\begin{equation}
  \mathrm{KL}(p \,\|\, q)
  \;=\; p \log\frac{p}{q} + (1-p)\log\frac{1-p}{1-q}
  \;=\; H(p, q) - H(p) \;\ge\; 0,
  \label{eq:kl}
\end{equation}
with equality iff $q = p$ (Gibbs' inequality;
Theory/information-theory).
\end{definition}

\begin{remark}[Estimation as information minimization]
The average negative log-likelihood of Section~5 satisfies
$\mathcal{L}(q) = H(\xbar, q) = H(\xbar) + \mathrm{KL}(\xbar \| q)$:
maximizing likelihood is minimizing the cross-entropy between the
empirical distribution and the model, hence the KL divergence ---
which \eqref{eq:kl} shows is achieved exactly at $q = \xbar$, a
one-line second proof of Theorem~\ref{thm:mle}.
\end{remark}

\begin{remark}[Jeffreys prior]
The information-geometric default prior
$\pi(p) \propto \sqrt{I(p)} = p^{-1/2}(1-p)^{-1/2}$ is
$\mathrm{Beta}(\tfrac12, \tfrac12)$: the invariant reference prior is
a member of the conjugate family of Section~3, tying the two sections
together.
\end{remark}

\section{Numerical solution by stochastic gradient descent}

Estimation problems beyond this one rarely admit closed forms; the
Bernoulli case serves as the minimal instance of the general
numerical route. Define the average negative log-likelihood
$\mathcal{L}(p) = -\tfrac1n \ell(p) = -\xbar \log p -
(1-\xbar)\log(1-p)$, whose minimizer coincides with \eqref{eq:mle}.
To remove the constraint $p \in (0,1)$, reparameterize through the
logistic function $\sigma(t) = (1 + e^{-t})^{-1}$ and minimize the
same objective in the new coordinate, $F(t) =
\mathcal{L}(\sigma(t))$, over $t \in \mathbb{R}$.

\begin{proposition}[Gradients]
\begin{equation}
  \boxed{\;F'(t) \;=\; \sigma(t) - \xbar,
  \qquad
  g_i(t) \;=\; \sigma(t) - x_i,\;}
  \label{eq:grad}
\end{equation}
where $g_i$ denotes the gradient of the $i$-th per-sample loss;
$\E_i[g_i(t)] = F'(t)$ for a uniformly drawn index, and
$F''(t) = \sigma'(t) > 0$, so $F$ is strictly convex.
\end{proposition}

\begin{proof}
First the derivative of the sigmoid itself. From
$\sigma(t) = (1 + e^{-t})^{-1}$,
\begin{equation}
  \sigma'(t)
  \;=\; \frac{e^{-t}}{(1 + e^{-t})^{2}}
  \;=\; \frac{1}{1 + e^{-t}} \cdot \frac{e^{-t}}{1 + e^{-t}}
  \;=\; \sigma(t)\bigl(1 - \sigma(t)\bigr),
  \label{eq:sigprime}
\end{equation}
using $1 - \sigma(t) = e^{-t}/(1 + e^{-t})$. Now the chain rule.
Writing $s = \sigma(t)$,
\begin{equation*}
  F'(t)
  \;=\; \mathcal{L}'(s)\,\sigma'(t)
  \;=\; \left[-\frac{\xbar}{s} + \frac{1-\xbar}{1-s}\right]
        s(1-s)
  \;=\; -\xbar(1-s) + (1-\xbar)\,s
  \;=\; s - \xbar,
\end{equation*}
every factor cancelling except the residual. The per-sample case is
identical with $x_i$ in place of $\xbar$, and taking
$\E_i$ over a uniform index averages $x_i$ back to $\xbar$.
Differentiating once more, $F''(t) = \sigma'(t) > 0$ by
\eqref{eq:sigprime}.
\end{proof}

Stochastic gradient descent iterates
$t \leftarrow t - \eta\, g_i(t)$ over shuffled passes through the
sample. With a constant step size $\eta$ the iterate reaches a
stationary distribution concentrated near the optimum rather than
converging; the estimator is therefore taken as the
Polyak--Ruppert average of the final pass, mapped back through
$\sigma$ (by the invariance of maximum likelihood,
$\phat = \sigma(\hat t\,)$). Automatic differentiation of the
logistic cross-entropy reproduces \eqref{eq:grad}; driving both
implementations through an identical sample schedule yields
identical trajectories, which the companion program asserts.
