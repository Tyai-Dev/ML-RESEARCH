\begin{abstract}
This note is a self-contained reference for the Bernoulli
distribution: its definition and elementary properties, the sampling
properties of its natural estimator, maximum likelihood and maximum a
posteriori estimation, and the numerical solution of the maximum
likelihood problem by stochastic gradient descent. All quantitative
statements are verified numerically by the companion program
\texttt{bernoulli.py}.
\end{abstract}

\section{The distribution}

\begin{definition}
A random variable $X$ follows the \emph{Bernoulli distribution} with
parameter $p \in [0,1]$, written $X \sim \mathrm{Bernoulli}(p)$, if
$X$ takes values in $\{0, 1\}$ with
\begin{equation}
  \Pr(X = 1) = p, \qquad \Pr(X = 0) = 1 - p.
\end{equation}
\end{definition}

The probability mass function admits the compact form
\begin{equation}
  f(x; p) \;=\; p^{x}(1-p)^{1-x}, \qquad x \in \{0, 1\},
  \label{eq:pmf}
\end{equation}
and the cumulative distribution function is the step function
\begin{equation}
  F(x; p) \;=\;
  \begin{cases}
    0, & x < 0,\\
    1 - p, & 0 \le x < 1,\\
    1, & x \ge 1.
  \end{cases}
  \label{eq:cdf}
\end{equation}

\section{Statistics of the estimator}

Let $x_1, \dots, x_n$ be an i.i.d.\ sample from
$\mathrm{Bernoulli}(p)$ and write $m = \sum_i x_i$ for the number of
successes.

\begin{proposition}[Sufficiency]
The likelihood depends on the data only through $m$:
\begin{equation}
  L(p) \;=\; \prod_{i=1}^{n} p^{x_i}(1-p)^{1-x_i}
        \;=\; p^{m}(1-p)^{n-m}.
  \label{eq:lik}
\end{equation}
By the factorization criterion, $m$ (equivalently $\xbar = m/n$) is a
sufficient statistic for $p$.
\end{proposition}

Throughout, $\phat = m/n$ denotes the sample mean; its sampling
properties follow from $m \sim \mathrm{Binomial}(n, p)$.

\begin{proposition}[Finite-sample properties]
\label{prop:finite}
For every $n \ge 1$,
\begin{equation}
  \E[\phat] = p,
  \qquad
  \Var(\phat) = \frac{p(1-p)}{n},
  \qquad
  \mathrm{MSE}(\phat) = \frac{p(1-p)}{n}.
\end{equation}
Moreover $\Var(\phat)$ attains the Cram\'er--Rao lower bound
$1/(n I(p))$: among unbiased estimators of $p$, the sample mean is
\emph{efficient} (see \textnormal{Theory/statistics} for the general
bound).
\end{proposition}

\begin{proposition}[Asymptotics]
\label{prop:asym}
$\phat \xrightarrow{\,a.s.\,} p$ (strong law of large numbers), and by
the central limit theorem
\begin{equation}
  \sqrt{n}\,(\phat - p) \;\xrightarrow{\ d\ }\;
  \mathcal{N}\bigl(0,\, p(1-p)\bigr).
  \label{eq:clt}
\end{equation}
\end{proposition}

\begin{corollary}[Confidence interval]
An asymptotic $95\%$ confidence interval for $p$ is
$\phat \pm 1.96\sqrt{\phat(1-\phat)/n}$.
\end{corollary}

\section{Maximum likelihood estimation}

\begin{theorem}[MLE]
\label{thm:mle}
Let $0 < m < n$. The unique maximizer of \eqref{eq:lik} over
$p \in (0,1)$ is
\begin{equation}
  \boxed{\;\phat_{\mathrm{MLE}} \;=\; \frac{m}{n} \;=\; \xbar.\;}
  \label{eq:mle}
\end{equation}
\end{theorem}

\begin{proof}
Since $\log$ is strictly increasing, maximizing $L$ is equivalent to
maximizing
\begin{equation}
  \ell(p) \;=\; \log L(p) \;=\; m \log p + (n - m)\log(1 - p).
  \label{eq:loglik}
\end{equation}
Differentiating and equating to zero,
\begin{equation*}
  \ell'(p) = \frac{m}{p} - \frac{n-m}{1-p} = 0
  \;\Longleftrightarrow\;
  m(1-p) = (n-m)\,p
  \;\Longleftrightarrow\;
  m = np,
\end{equation*}
which yields \eqref{eq:mle}. Uniqueness follows from strict concavity:
$\ell''(p) = -m/p^{2} - (n-m)/(1-p)^{2} < 0$ on $(0,1)$.
\end{proof}

\begin{remark}[Boundary cases]
If $m = 0$ or $m = n$, the likelihood is monotone on $(0,1)$ and its
supremum is attained at the boundary point $m/n$; formula
\eqref{eq:mle} remains valid.
\end{remark}

\section{Maximum a posteriori estimation}

\begin{proposition}[Conjugacy]
Let $p \sim \mathrm{Beta}(\alpha, \beta)$, i.e.\
$\pi(p) \propto p^{\alpha-1}(1-p)^{\beta-1}$. Then the posterior given
the sample is again Beta:
\begin{equation}
  \pi(p \mid x_{1:n}) \;\propto\; L(p)\,\pi(p)
  \;\propto\; p^{\,m+\alpha-1}(1-p)^{\,n-m+\beta-1}
  \;=\; \mathrm{Beta}(m + \alpha,\; n - m + \beta).
\end{equation}
\end{proposition}

\begin{theorem}[MAP]
For $m + \alpha > 1$ and $n - m + \beta > 1$, the posterior mode is
\begin{equation}
  \boxed{\;\phat_{\mathrm{MAP}}
  \;=\; \frac{m + \alpha - 1}{\,n + \alpha + \beta - 2\,}.\;}
  \label{eq:map}
\end{equation}
\end{theorem}

\begin{proof}
The posterior has the functional form of a likelihood
\eqref{eq:lik} with counts $m' = m + \alpha - 1$ and
$n' = n + \alpha + \beta - 2$; apply Theorem~\ref{thm:mle}.
\end{proof}

\begin{remark}[Interpretation and limits]
The prior acts as \emph{pseudo-counts}: $\alpha - 1$ successes and
$\beta - 1$ failures observed before any data. In particular
(i) the flat prior $\alpha = \beta = 1$ gives
$\phat_{\mathrm{MAP}} = \phat_{\mathrm{MLE}}$ exactly;
(ii) $\phat_{\mathrm{MAP}} \to \phat_{\mathrm{MLE}}$ as
$n \to \infty$; and (iii) for small $n$ the estimator shrinks toward
the prior mode, trading bias for variance --- numerically, at
$n = 20$ with $\mathrm{Beta}(3,3)$ the MAP mean squared error is
$0.0085$ against the MLE's $0.0105$ ($2 \times 10^4$ replications).
Additive (Laplace) smoothing of categorical counts is precisely the
multinoulli analog of \eqref{eq:map} under a Dirichlet prior.
\end{remark}

\section{Numerical solution by stochastic gradient descent}

Estimation problems beyond this one rarely admit closed forms; the
Bernoulli case serves as the minimal instance of the general
numerical route. Define the average negative log-likelihood
$\mathcal{L}(p) = -\tfrac1n \ell(p) = -\xbar \log p -
(1-\xbar)\log(1-p)$, whose minimizer coincides with \eqref{eq:mle}.
To remove the constraint $p \in (0,1)$, reparameterize through the
logistic function $\sigma(t) = (1 + e^{-t})^{-1}$ and minimize the
same objective in the new coordinate, $F(t) =
\mathcal{L}(\sigma(t))$, over $t \in \mathbb{R}$.

\begin{proposition}[Gradients]
With $\sigma'(t) = \sigma(t)(1 - \sigma(t))$,
\begin{equation}
  \boxed{\;F'(t) \;=\; \sigma(t) - \xbar,
  \qquad
  g_i(t) \;=\; \sigma(t) - x_i,\;}
  \label{eq:grad}
\end{equation}
where $g_i$ denotes the gradient of the $i$-th per-sample loss;
$\E_i[g_i(t)] = F'(t)$ for a uniformly drawn index, and
$F''(t) = \sigma'(t) > 0$, so $F$ is strictly convex.
\end{proposition}

Stochastic gradient descent iterates
$t \leftarrow t - \eta\, g_i(t)$ over shuffled passes through the
sample. With a constant step size $\eta$ the iterate reaches a
stationary distribution concentrated near the optimum rather than
converging; the estimator is therefore taken as the
Polyak--Ruppert average of the final pass, mapped back through
$\sigma$ (by the invariance of maximum likelihood,
$\phat = \sigma(\hat t\,)$). Automatic differentiation of the
logistic cross-entropy reproduces \eqref{eq:grad}; driving both
implementations through an identical sample schedule yields
identical trajectories, which the companion program asserts.
